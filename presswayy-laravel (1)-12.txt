<?php

namespace Presswayy\Sync\Console;

use Illuminate\Console\Command;
use Presswayy\Sync\Client;
use Presswayy\Sync\Jobs\BulkSyncPage;

class SyncCommand extends Command
{
protected $signature = 'presswayy:sync';
protected $description = 'Queue a full catalog sync to Presswayy';

public function handle()
{
if (!Client::isConfigured()) {
$this->error('Set PRESSWAYY_KEY in your .env first.');
return 1;
}
BulkSyncPage::dispatch(1);
$this->info('Full sync queued. Make sure a queue worker is running.');
return 0;
}
}
