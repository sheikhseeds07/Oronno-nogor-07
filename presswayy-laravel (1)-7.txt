<?php

namespace Presswayy\Sync\Http;

use Illuminate\Http\Request;
use Illuminate\Routing\Controller;
use Presswayy\Sync\Contracts\StoreAdapter;
use Presswayy\Sync\Jobs\BulkSyncPage;

/**
* Inbound endpoints the dashboard calls (dashboard -> your store). All requests
* are HMAC-verified by the VerifySignature middleware before reaching here.
*/
class InboundController extends Controller
{
public function order(Request $request, StoreAdapter $adapter)
{
$data = $request->json()->all();
if (empty($data['line_items']) || !is_array($data['line_items'])) {
return response()->json(['ok' => false, 'error' => 'no_line_items'], 400);
}
try {
$result = $adapter->createOrder($data);
} catch (\Throwable $e) {
return response()->json(['ok' => false, 'error' => $e->getMessage()], 422);
}
return response()->json(array_merge(['ok' => true], $result), 200);
}

public function inventory(Request $request, StoreAdapter $adapter)
{
$data = $request->json()->all();
$variationId = !empty($data['variation_id']) ? (string) $data['variation_id'] : null;
$productId = isset($data['product_id']) ? (string) $data['product_id'] : null;
if (!isset($data['stock_quantity']) || (!$productId && !$variationId)) {
return response()->json(['ok' => false, 'error' => 'missing_fields'], 400);
}
try {
$adapter->setStock($productId, $variationId, (int) $data['stock_quantity']);
} catch (\Throwable $e) {
return response()->json(['ok' => false, 'error' => $e->getMessage()], 422);
}
return response()->json(['ok' => true, 'stock_quantity' => (int) $data['stock_quantity']], 200);
}

public function resync()
{
BulkSyncPage::dispatch(1);
return response()->json(['ok' => true, 'message' => 'Resync started'], 202);
}

public function ping()
{
return response()->json([
'ok'               => true,
'platform'         => 'laravel',
'platform_version' => app()->version(),
'store_url'        => rtrim((string) config('app.url'), '/'),
], 200);
}
}
