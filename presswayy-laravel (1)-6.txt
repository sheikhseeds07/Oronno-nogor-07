<?php

namespace Presswayy\Sync\Adapters;

use Presswayy\Sync\Contracts\StoreAdapter;

/**
* Config-driven adapter — maps your models to Presswayy's canonical shape using
* the column mappings in config/presswayy.php. Covers the read/push side and
* stock write-back for the common case with no code. Order write-back needs a
* custom adapter (createOrder throws here) because creating an order touches
* app-specific business logic.
*/
class GenericAdapter implements StoreAdapter
{
/** Read a mapped column off a model, or a default when unmapped. */
private function col($model, $group, $key, $default = null)
{
$columns = config("presswayy.$group.columns", []);
if (!empty($columns[$key])) {
return data_get($model, $columns[$key]);
}
return $default;
}

public function product($model)
{
if (!$model) {
return null;
}
return [
'id'             => $this->col($model, 'product', 'id') ?: $model->getKey(),
'name'           => $this->col($model, 'product', 'name'),
'description'    => $this->col($model, 'product', 'description'),
'sku'            => $this->col($model, 'product', 'sku'),
'regular_price'  => $this->col($model, 'product', 'regular_price'),
'sale_price'     => $this->col($model, 'product', 'sale_price'),
'stock_quantity' => $this->col($model, 'product', 'stock_quantity'),
'status'         => $this->col($model, 'product', 'status', 'publish'),
'slug'           => $this->col($model, 'product', 'slug'),
'image_url'      => $this->col($model, 'product', 'image_url'),
'gallery_images' => [],
'categories'     => [],
'variations'     => [], // generic adapter handles simple products
];
}

public function order($model)
{
if (!$model) {
return null;
}
$items = [];
$relation = config('presswayy.order.line_items_relation');
$lic = config('presswayy.order.line_item_columns', []);
if ($relation && isset($model->{$relation})) {
foreach ($model->{$relation} as $li) {
$items[] = [
'product_id'   => !empty($lic['product_id']) ? data_get($li, $lic['product_id']) : null,
'variation_id' => !empty($lic['variation_id']) ? data_get($li, $lic['variation_id']) : null,
'name'         => !empty($lic['name']) ? data_get($li, $lic['name']) : null,
'quantity'     => !empty($lic['quantity']) ? data_get($li, $lic['quantity']) : 1,
'total'        => !empty($lic['total']) ? data_get($li, $lic['total']) : null,
'sku'          => !empty($lic['sku']) ? data_get($li, $lic['sku']) : null,
];
}
}
$createdViaCol = config('presswayy.order.columns.created_via');
$createdVia = $createdViaCol ? data_get($model, $createdViaCol) : null;

return [
'id'            => $this->col($model, 'order', 'id') ?: $model->getKey(),
'number'        => $this->col($model, 'order', 'number', $model->getKey()),
'status'        => $this->col($model, 'order', 'status'),
'total'         => $this->col($model, 'order', 'total'),
'currency'      => $this->col($model, 'order', 'currency'),
'customer_id'   => $this->col($model, 'order', 'customer_id'),
'email'         => $this->col($model, 'order', 'email'),
'phone'         => $this->col($model, 'order', 'phone'),
'billing'       => [
'first_name' => $this->col($model, 'order', 'first_name'),
'last_name'  => $this->col($model, 'order', 'last_name'),
'email'      => $this->col($model, 'order', 'email'),
'phone'      => $this->col($model, 'order', 'phone'),
'address_1'  => $this->col($model, 'order', 'address_1'),
],
'line_items'    => $items,
'created_via'   => $createdVia ? 'presswayy-ai' : null,
];
}

public function customer($model)
{
if (!$model) {
return null;
}
return [
'id'         => $this->col($model, 'customer', 'id') ?: $model->getKey(),
'first_name' => $this->col($model, 'customer', 'first_name'),
'last_name'  => $this->col($model, 'customer', 'last_name'),
'name'       => $this->col($model, 'customer', 'name'),
'email'      => $this->col($model, 'customer', 'email'),
'phone'      => $this->col($model, 'customer', 'phone'),
];
}

public function allProducts($page, $perPage)
{
$model = config('presswayy.product.model');
if (!$model) {
return [];
}
return $model::query()->forPage($page, $perPage)->get();
}

public function observedModels()
{
return array_filter([
'product'  => config('presswayy.product.model'),
'order'    => config('presswayy.order.model'),
'customer' => config('presswayy.customer.model'),
]);
}

public function createOrder(array $payload)
{
throw new \RuntimeException(
'The generic adapter cannot create orders. Implement ' .
'Presswayy\\Sync\\Contracts\\StoreAdapter::createOrder() in a custom adapter ' .
'for order write-back, then set PRESSWAYY_ADAPTER accordingly.'
);
}

public function setStock($productId, $variationId, $quantity)
{
$model = config('presswayy.product.model');
$idCol = config('presswayy.product.columns.id', 'id');
$stockCol = config('presswayy.product.columns.stock_quantity');
if (!$model || !$stockCol) {
throw new \RuntimeException('Generic adapter: product model or stock column not configured.');
}
$id = $variationId ? $variationId : $productId;
$model::query()->where($idCol, $id)->update([$stockCol => (int) $quantity]);
}
}
