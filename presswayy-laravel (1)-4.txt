<?php

namespace Presswayy\Sync\Contracts;

/**
* Maps YOUR store's models to Presswayy's canonical shape, and fulfils writes
* back from the dashboard. Implement this for a fully-custom schema, or use the
* config-driven GenericAdapter (adapter=generic).
*/
interface StoreAdapter
{
/** @return array|null canonical product payload */
public function product($model);

/** @return array|null canonical order payload */
public function order($model);

/** @return array|null canonical customer payload */
public function customer($model);

/** @return iterable a page of product models for bulk sync */
public function allProducts($page, $perPage);

/** Which models to observe: ['product'=>Class, 'order'=>Class, 'customer'=>Class]. */
public function observedModels();

/**
* Create an order in your store from a dashboard AI-chat order.
* @return array ['order_id'=>..,'order_number'=>..]
*/
public function createOrder(array $payload);

/** Set stock for a product (or variation) in your store. */
public function setStock($productId, $variationId, $quantity);
}
