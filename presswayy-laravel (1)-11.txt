<?php

namespace Presswayy\Sync;

use Illuminate\Support\Facades\Http;

/**
* HMAC transport to the Presswayy dashboard. The connection key is
* "<id>.<secret>": the id is public (identifies the store), the secret signs
* every request body — base64( HMAC-SHA256(rawBody, secret) ), byte-identical
* to the dashboard's verification.
*/
class Client
{
public static function key()
{
return (string) config('presswayy.key', '');
}

public static function isConfigured()
{
$k = self::key();
return $k !== '' && strpos($k, '.') !== false;
}

/** @return array{0:string,1:string} [id, secret] */
public static function parts()
{
$k = self::key();
if (!self::isConfigured()) {
return ['', ''];
}
$p = explode('.', $k, 2);
return [$p[0], isset($p[1]) ? $p[1] : ''];
}

public static function apiBase()
{
return rtrim((string) config('presswayy.api_base', 'https://app.presswayy.com'), '/');
}

public static function sign($body)
{
$parts = self::parts();
return base64_encode(hash_hmac('sha256', $body, $parts[1], true));
}

/** Verify an inbound (dashboard -> store) signature over the raw body. */
public static function verify($rawBody, $signature)
{
$parts = self::parts();
if ($parts[1] === '' || !$signature) {
return false;
}
$expected = base64_encode(hash_hmac('sha256', $rawBody, $parts[1], true));
return hash_equals($expected, (string) $signature);
}

/**
* Send a signed webhook to the dashboard.
* @return array{ok:bool,status:int,body?:mixed,error?:string}
*/
public function send($topic, $payload, $blocking = false)
{
$parts = self::parts();
$id = $parts[0];
$secret = $parts[1];
if ($id === '' || $secret === '') {
return ['ok' => false, 'status' => 0, 'error' => 'not_configured'];
}

$body = json_encode($payload);
$sig = base64_encode(hash_hmac('sha256', $body, $secret, true));

$response = Http::withHeaders([
'Content-Type'          => 'application/json',
'X-Presswayy-Key'       => $id,
'X-Presswayy-Topic'     => $topic,
'X-Presswayy-Signature' => $sig,
])->withBody($body, 'application/json')
->timeout($blocking ? 25 : 15)
->post(self::apiBase() . '/api/store/webhooks');

return [
'ok'     => $response->successful(),
'status' => $response->status(),
'body'   => $response->json(),
];
}

/** Register this store with the dashboard (blocking). */
public function register()
{
$payload = [
'store_url'        => rtrim((string) config('app.url'), '/'),
'platform'         => 'laravel',
'framework'        => config('presswayy.adapter', 'generic'),
'inbound_base'     => config('presswayy.inbound_base', '/presswayy/v1'),
'platform_version' => app()->version(),
'currency'         => config('presswayy.currency'),
];
return $this->send('register', $payload, true);
}
}
